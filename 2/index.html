1.1: Convolutions from Scratch

convolution with 4 for loops: 


        
def four_loop_convolution(arr, kernel):
    m1, n1 = arr.shape
    m2, n2 = kernel.shape
    px, py = m2//2, n2//2
    padded_arr = np.pad(arr, ((px, px), (py, py)), 'constant')
    res = np.zeros_like(arr)

    for i1 in range(m1):
    for j1 in range(n1):
        total = 0.0
        for i2 in range(m2):
        for j2 in range(n2):
            total += padded_arr[i1+i2][j1+j2] * kernel[i2][j2]
        res[i1][j1] = total
    return res

(better) convolution with 2 for loops: 


def convolve(arr, kernel):
    m1, n1 = arr.shape
    m2, n2 = kernel.shape
    px, py = m2//2, n2//2
    padded_arr = np.pad(arr, ((px, px), (py, py)), 'constant')
    res = np.zeros_like(arr)
    
    for i1 in range(m1):
        for j1 in range(n1):
        res[i1][j1] = np.sum(padded_arr[i1:i1+m2, j1:j1+n2] * kernel)
    
    return res
